  solution={"#include <string>\n#include <string_view>\n#include <vector>\n#include <unordered_map>\n#include <algorithm>\nstd::vector<std::pair<std::string, int>> top_words(std::string_view text, std::size_t k) {\n    std::unordered_map<std::string, int> counts;\n    std::size_t i = 0;\n    while (i < text.size()) {\n        std::size_t j = text.find(' ', i);\n        if (j == std::string_view::npos) j = text.size();\n        if (j > i) ++counts[std::string(text.substr(i, j - i))];\n        i = j + 1;\n    }\n    std::vector<std::pair<std::string, int>> v(counts.begin(), counts.end());\n    std::sort(v.begin(), v.end(), [](const auto& a, const auto& b) { return a.second != b.second ? a.second > b.second : a.first < b.first; });\n    if (v.size() > k) v.resize(k);\n    return v;\n}\n"}
