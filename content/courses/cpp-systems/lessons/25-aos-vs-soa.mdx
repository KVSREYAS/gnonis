## Convert it

Sources: [Mike Acton, Data-Oriented Design and C++ (CppCon 2014)](https://www.youtube.com/watch?v=rX0ItVEVjHc), Richard Fabian, [*Data-Oriented Design*](https://www.dataorienteddesign.com/dodbook/)

  id="aos-to-soa"
  devices={["desktop"]}

  prompt={"Given `struct P { float x, y, z, mass; };`, implement `struct Soa { std::vector<float> x, y, z, mass; }`, `Soa to_soa(const std::vector<P>&)` and `float total_mass_above(const Soa&, float y_min)`, which sums mass where y > y_min."}
  starter={"#include <vector>\nstruct P { float x, y, z, mass; };\nstruct Soa { std::vector<float> x, y, z, mass; };\nSoa to_soa(const std::vector<P>& ps) { return {}; }\nfloat total_mass_above(const Soa& s, float y_min) { return 0; }\n"}

  tests={"#include \"check.h\"\n\nint main() {\n    std::vector<P> ps;\n    for (int i = 0; i < 1000; ++i) ps.push_back({float(i), float(i % 10), float(-i), 1.0f + (i % 3)});\n    auto s = to_soa(ps);\n    CHECK(s.x.size() == 1000 && s.z[999] == -999 && s.mass[2] == 3);\n    float expect = 0; for (auto& p : ps) if (p.y > 4.5f) expect += p.mass;\n    CHECK(total_mass_above(s, 4.5f) == expect);\n    CHECK(total_mass_above(to_soa({}), 0) == 0);\n    return 0;\n}\n"}
  solution={"#include <vector>\nstruct P { float x, y, z, mass; };\nstruct Soa { std::vector<float> x, y, z, mass; };\nSoa to_soa(const std::vector<P>& ps) {\n    Soa s; auto n = ps.size();\n    s.x.reserve(n); s.y.reserve(n); s.z.reserve(n); s.mass.reserve(n);\n    for (const auto& p : ps) { s.x.push_back(p.x); s.y.push_back(p.y); s.z.push_back(p.z); s.mass.push_back(p.mass); }\n    return s;\n}\nfloat total_mass_above(const Soa& s, float y_min) {\n    float t = 0;\n    for (std::size_t i = 0; i < s.y.size(); ++i) t += s.y[i] > y_min ? s.mass[i] : 0.f;\n    return t;\n}\n"}
  hints={["reserve each vector once.", "A ternary in the loop keeps it branch-free."]}
  explanation="An LLM's KV cache is SoA by design: all keys for a layer are contiguous."
