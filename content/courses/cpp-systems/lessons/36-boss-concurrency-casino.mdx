## Getaway: bounded blocking queue

Producers wait when the table is full. Consumers wait when it is empty.

  solution={"#include <mutex>\n#include <condition_variable>\n#include <deque>\ntemplate<class T> class BoundedQueue {\n    std::mutex m_; std::condition_variable not_full_, not_empty_; std::deque<T> q_; std::size_t cap_;\npublic:\n    explicit BoundedQueue(std::size_t cap) : cap_(cap) {}\n    void push(T v) {\n        { std::unique_lock lk(m_); not_full_.wait(lk, [&] { return q_.size() < cap_; }); q_.push_back(std::move(v)); }\n        not_empty_.notify_one();\n    }\n    T pop() {\n        std::unique_lock lk(m_); not_empty_.wait(lk, [&] { return !q_.empty(); });\n        T v = std::move(q_.front()); q_.pop_front(); lk.unlock(); not_full_.notify_one(); return v;\n    }\n    bool try_pop(T& out) {\n        std::unique_lock lk(m_); if (q_.empty()) return false;\n        out = std::move(q_.front()); q_.pop_front(); lk.unlock(); not_full_.notify_one(); return true;\n    }\n};\n"}
