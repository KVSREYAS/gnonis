  - "An arena owns one big buffer. Allocation is align_up(offset) then bump, which is O(1) with no per-object free."
  - "Reset frees everything at once, which suits per-request or per-step lifetimes."
  - "std::pmr::monotonic_buffer_resource is the standard arena and plugs into std::pmr containers."

  visibleTests={"#include \"check.h\"\n\nint main() { Arena a(256); CHECK(a.alloc(8, 8) != nullptr && a.used() >= 8); return 0; }\n"}
  tests={"#include \"check.h\"\n\n#include <cstdint>\n#include <type_traits>\nint main() {\n    static_assert(!std::is_copy_constructible_v<Arena>);\n    Arena a(1024);\n    auto* f = a.make_array<float>(10); CHECK(f && f[9] == 0.f);\n    void* p = a.alloc(1, 64); CHECK(p && reinterpret_cast<std::uintptr_t>(p) % 64 == 0);\n    CHECK(a.alloc(4096, 8) == nullptr);\n    std::size_t u = a.used(); CHECK(u > 40);\n    a.reset(); CHECK(a.used() == 0);\n    auto* d = a.make_array<double>(128); CHECK(d && d[127] == 0.0);\n    CHECK(a.make_array<double>(1) == nullptr);\n    return 0;\n}\n"}
  solution={"#include <cstddef>\n#include <cstdint>\n#include <memory>\n#include <new>\n#include <type_traits>\nclass Arena {\n    std::unique_ptr<std::byte[]> buf_; std::size_t cap_, off_ = 0;\npublic:\n    explicit Arena(std::size_t bytes) : buf_(new std::byte[bytes]), cap_(bytes) {}\n    Arena(const Arena&) = delete; Arena& operator=(const Arena&) = delete;\n    void* alloc(std::size_t n, std::size_t align) {\n        auto base = reinterpret_cast<std::uintptr_t>(buf_.get());\n        auto p = (base + off_ + align - 1) & ~(std::uintptr_t)(align - 1);\n        if (p + n > base + cap_) return nullptr;\n        off_ = p + n - base; return reinterpret_cast<void*>(p);\n    }\n    template<class T> T* make_array(std::size_t n) {\n        static_assert(std::is_trivially_default_constructible_v<T>);\n        void* p = alloc(n * sizeof(T), alignof(T)); if (!p) return nullptr;\n        return new (p) T[n]();\n    }\n    void reset() { off_ = 0; }\n    std::size_t used() const { return off_; }\n};\n"}
