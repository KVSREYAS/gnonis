  solution={"#include <vector>\n#include <memory>\n#include <initializer_list>\n#include <cstddef>\n#include <utility>\nstruct Tensor {\n    std::shared_ptr<std::vector<float>> data; std::vector<std::size_t> shape, stride; std::size_t offset = 0;\n    static Tensor zeros(std::vector<std::size_t> shape) {\n        Tensor t; t.shape = shape; t.stride.assign(shape.size(), 1);\n        std::size_t n = 1;\n        for (std::size_t i = shape.size(); i-- > 0;) { t.stride[i] = n; n *= shape[i]; }\n        t.data = std::make_shared<std::vector<float>>(n, 0.f); return t;\n    }\n    float& at(std::initializer_list<std::size_t> idx) {\n        std::size_t o = offset, d = 0; for (auto i : idx) o += i * stride[d++];\n        return (*data)[o];\n    }\n    Tensor transpose(std::size_t a, std::size_t b) const { Tensor t = *this; std::swap(t.shape[a], t.shape[b]); std::swap(t.stride[a], t.stride[b]); return t; }\n    bool is_contiguous() const {\n        std::size_t n = 1;\n        for (std::size_t i = shape.size(); i-- > 0;) { if (shape[i] != 1 && stride[i] != n) return false; n *= shape[i]; }\n        return true;\n    }\n};\n"}
