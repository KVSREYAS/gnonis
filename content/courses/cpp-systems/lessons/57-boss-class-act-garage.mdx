## Getaway: the vehicle roster

Build a small class hierarchy and a roster that owns vehicles through base pointers.

  id="roster"
  devices={["desktop"]}
  language="cpp"
  prompt="Write an abstract `class Vehicle` with constructor `Vehicle(std::string name)`, `const std::string& name() const`, a pure virtual `int wheels() const` and a virtual destructor.   tests={"#include \"check.h\"\nint main() {\n    Roster r; CHECK(r.size() == 0 && r.total_wheels() == 0);\n    r.add(std::make_unique<Bike>(\"BMX\"));\n    r.add(std::make_unique<Car>(\"Greenwood\"));\n    r.add(std::make_unique<Car>(\"Savanna\"));\n    CHECK(r.size() == 3 && r.total_wheels() == 10);\n    Car c(\"Admiral\"); const Vehicle& v = c;\n    CHECK(v.name() == \"Admiral\" && v.wheels() == 4);\n    return 0;\n}\n"}
  solution={"#include <string>\n#include <memory>\n#include <vector>\n#include <utility>\nclass Vehicle {\n    std::string name_;\npublic:\n    Vehicle(std::string name) : name_(std::move(name)) {}\n    const std::string& name() const { return name_; }\n    virtual int wheels() const = 0;\n    virtual ~Vehicle() = default;\n};\nclass Bike : public Vehicle {\npublic:\n    Bike(std::string n) : Vehicle(std::move(n)) {}\n    int wheels() const override { return 2; }\n};\nclass Car : public Vehicle {\npublic:\n    Car(std::string n) : Vehicle(std::move(n)) {}\n    int wheels() const override { return 4; }\n};\nclass Roster {\n    std::vector<std::unique_ptr<Vehicle>> v_;\npublic:\n    void add(std::unique_ptr<Vehicle> v) { v_.push_back(std::move(v)); }\n    int total_wheels() const { int t = 0; for (const auto& v : v_) t += v->wheels(); return t; }\n    std::size_t size() const { return v_.size(); }\n};\n"}
