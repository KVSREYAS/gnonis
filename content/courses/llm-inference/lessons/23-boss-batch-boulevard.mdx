  solution={"def continuous_steps(lengths, slots):\n    queue = list(lengths)\n    active = []\n    steps = 0\n    while queue or active:\n        while len(active) < slots and queue:\n            active.append(queue.pop(0))\n        steps += 1\n        active = [r - 1 for r in active if r > 1]\n    return steps\n"}
