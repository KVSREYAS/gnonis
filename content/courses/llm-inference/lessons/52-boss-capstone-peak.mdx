  solution={"def simulate(lengths, slots):\n    queue = list(enumerate(lengths))\n    active = []\n    finish = [0] * len(lengths)\n    t = 0\n    while queue or active:\n        while len(active) < slots and queue:\n            i, n = queue.pop(0)\n            active.append([i, n])\n        t += 1\n        for r in active:\n            r[1] -= 1\n            if r[1] == 0:\n                finish[r[0]] = t\n        active = [r for r in active if r[1] > 0]\n    return finish\n"}
